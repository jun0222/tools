08Tabs

activeで黒背景・黄テキスト。シャープなセグメント。

HTML
CSS
JS
<div class="nb-tabs">
  <button class="nb-tab active" onclick="activateTab(this)">概要</button>
  <button class="nb-tab" onclick="activateTab(this)">統計</button>
  <button class="nb-tab" onclick="activateTab(this)">設定</button>
</div>
.nb-tabs {
  display: flex;
  border: 2.5px solid #0a0a0a;
  border-radius: 4px;
  box-shadow: 4px 4px 0 #0a0a0a;
  overflow: hidden;
}
.nb-tab {
  flex: 1;
  padding: 10px 16px;
  border: none;
  border-right: 2.5px solid #0a0a0a;
  font-size: 0.78rem;
  font-weight: 800;
  background: #fff;
  cursor: pointer;
  text-transform: uppercase;
  letter-spacing: 0.04em;
}
.nb-tab:last-child { border-right: none; }
.nb-tab.active { background: #0a0a0a; color: #ffe000; }
function activateTab(el) {
  el.closest('.nb-tabs')
    .querySelectorAll('.nb-tab')
    .forEach(t => t.classList.remove('active'));
  el.classList.add('active');
}
