Tabs

ピル型のセグメントタブ。activeクラスをJSで付け替えて使う。

HTML
CSS
JS
<div class="nb-tabs">
  <button class="nb-tab active" onclick="activateTab(this)">概要</button>
  <button class="nb-tab" onclick="activateTab(this)">統計</button>
  <button class="nb-tab" onclick="activateTab(this)">設定</button>
</div>
.nb-tabs {
  display: flex;
  gap: 2px;
  background: #f1f5f9;
  padding: 4px;
  border-radius: 14px;
}
.nb-tab {
  flex: 1;
  padding: 9px 16px;
  border: none;
  border-radius: 10px;
  font-size: 0.82rem;
  font-weight: 700;
  color: #64748b;
  background: transparent;
  cursor: pointer;
  transition: color 0.2s, background 0.2s, box-shadow 0.2s;
}
.nb-tab.active {
  background: white;
  color: #3b82f6;
  box-shadow: 0 2px 8px rgba(0,0,0,0.08);
}
function activateTab(el) {
  // 同じ .nb-tabs 内のタブを全部非アクティブに
  el.closest('.nb-tabs')
    .querySelectorAll('.nb-tab')
    .forEach(t => t.classList.remove('active'));
  el.classList.add('active');
}
